train_step.py
model = nn.Sequential(nn.Linear(3, 8), nn.ReLU(), nn.Linear(8, 2))
opt, criterion = torch.optim.AdamW(model.parameters(), lr=1e-2), nn.CrossEntropyLoss()

model.train()
opt.zero_grad()                  # 1. reset grads
loss = criterion(model(x), y)    # 2. forward + scalar loss (e.g., 0.71)
loss.backward()                  # 3. compute grads
opt.step()                       # 4. update weights
